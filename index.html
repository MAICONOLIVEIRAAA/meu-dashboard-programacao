// --- LÓGICA DA FOTO --- function abrirModal() { document.getElementById('modal-foto').classList.remove('hidden'); } function fecharModal() { document.getElementById('modal-foto').classList.add('hidden'); } function salvarFoto(url) { try { localStorage.setItem('userAvatar', url); document.getElementById('profile-img').src = url; document.getElementById('profile-img-small').src = url; fecharModal(); } catch (e) { alert("Erro: A imagem é muito pesada para ser salva no navegador. Tente uma foto menor ou com qualidade mais baixa."); console.error(e); } } function uploadFoto(input) { if (input.files && input.files[0]) { const arquivo = input.files[0]; // LIMITE DE TAMANHO (2MB) const limiteTamanho = 2 * 1024 * 1024; // 2MB if (arquivo.size > limiteTamanho) { alert("A imagem é muito grande! Por favor, escolha uma foto menor que 2MB."); input.value = ""; // Limpa a seleção return; } const reader = new FileReader(); reader.onload = (e) => { salvarFoto(e.target.result); }; reader.readAsDataURL(arquivo); } } function gerarAvatar() { salvarFoto(`https://api.dicebear.com/7.x/avataaars/svg?seed=${Math.random()}`); }
